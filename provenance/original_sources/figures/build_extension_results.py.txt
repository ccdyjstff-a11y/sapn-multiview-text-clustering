def name(method,zh=False):
    if method in SHORT:return SHORT[method][int(zh)]
    if method=='MVCHSS':return r'MVCHSS$^\dagger$'
    if method=='FSSMSC':return r'FSSMSC$^\ddagger$'
    return method


def dsname(ds):return DATASET_NAMES[ds]['en']


def pm(mean,sd):
    return '--' if mean is None else f'${mean:.2f}\\pm {sd:.2f}$' if sd is not None else f'${mean:.2f}$'


def signed(x):return f'{x:+.2f}'


def longtable(header,rows,caption,label,columns,group_size=None):
    h=' & '.join(header)+r'\\'
    return '\n'.join([r'\begingroup\small',r'\setlength{\tabcolsep}{4pt}',
        r'\begin{longtable}{'+columns+'}',r'\caption{'+caption+r'}\label{'+label+r'}\\',
        r'\toprule',h,r'\midrule\endfirsthead',r'\toprule',h,r'\midrule\endhead',
        r'\bottomrule\endfoot',*[' & '.join(map(str,row))+
            (r'\\*' if group_size and (i+1)%group_size else r'\\')
            for i,row in enumerate(rows)],
        r'\end{longtable}',r'\endgroup'])


def compact_table(header,rows,caption,label,columns):
    """Keep the small diagnostic table intact; numerical contents are unchanged."""
    return '\n'.join([r'\begin{table}[htbp]\centering\small',
        r'\setlength{\tabcolsep}{4pt}',r'\caption{'+caption+r'}\label{'+label+'}',
        r'\begin{tabular}{'+columns+r'}\toprule',' & '.join(header)+r'\\',
        r'\midrule',*[' & '.join(map(str,row))+r'\\' for row in rows],
        r'\bottomrule\end{tabular}',r'\end{table}'])


def cell_map(group):return {(c['dataset'],c['Q'],c['method']):c for c in group['cells']}
def pair_map(group):return {(c['dataset'],c['Q'],c['comparison']):c for c in group['paired']}


def absolute_table(group,datasets,methods,budget,zh,label,caption):
    lookup=cell_map(group);rows=[]
    for ds in datasets:
        for i,m in enumerate(methods):
            c=lookup[(ds,budget,m)]
            rows.append([dsname(ds) if i==0 else '',name(m,zh),
                *[pm(c['metrics'][metric]['primary_mean_percent'],c['metrics'][metric]['primary_sd_percent']) for metric in CLUSTER_METRICS],
                f"{c['eligible']}/{c['planned']}"])
    return longtable(['数据' if zh else 'Corpus','方法' if zh else 'Method','ACC','NMI','ARI','合格' if zh else 'Eligible'],rows,caption,label,'llrrrr',group_size=len(methods))


def paired_table(group,methods,budgets,zh,label,caption,direction=1):
    lookup=pair_map(group);rows=[]
    for ds in DEVELOPMENT:
        for Q in budgets:
            for method in methods:
                c=lookup[(ds,Q,method)]
                values=[]
                for metric in CLUSTER_METRICS:
                    m=c['metrics'][metric]
                    values.append(pm(None if m['mean_pp'] is None else direction*m['mean_pp'],m['sd_pp']))
                rows.append([dsname(ds),Q,name(method,zh),*values,f"{c['paired_eligible']}/{c['planned']}"])
    return longtable(['数据' if zh else 'Corpus','$Q$','对照' if zh else 'Control',
        '$\Delta$ACC','$\Delta$NMI','$\Delta$ARI','合格配对' if zh else 'Eligible pairs'],rows,caption,label,'lrlrrrr')


def effect_summary(group,method,Q,metric='ACC',direction=1):
    items=[p for p in group['paired'] if p['comparison']==method and p['Q']==Q]
    full=[(p['dataset'],direction*p['metrics'][metric]['mean_pp']) for p in items if p['metrics'][metric]['primary_eligible']]
    return dict(planned=len(items),complete=len(full),positive=sum(v>1e-10 for _,v in full),
        negative=sum(v < -1e-10 for _,v in full),ties=sum(abs(v)<=1e-10 for _,v in full),
        lower=min((v for _,v in full),default=None),upper=max((v for _,v in full),default=None),values=full)


def effects_range(e,zh=False):
    if not e['complete']:return '无完整单元' if zh else 'no complete cells'
    return f"{signed(e['lower'])} 至 {signed(e['upper'])}" if zh else f"{signed(e['lower'])} to {signed(e['upper'])}"


def budget_effects(group,methods=('SAPN-MV',),low=0,high=400):
    records=group['records'];lookup={(r['dataset'],r['seed'],r['Q'],r['method']):r for r in records};out=[]
    for ds,method in sorted({(r['dataset'],r['method']) for r in records if r['method'] in methods}):
        seeds=sorted({r['seed'] for r in records if (r['dataset'],r['method'])==(ds,method)})
        pairs=[(lookup[(ds,s,low,method)],lookup[(ds,s,high,method)]) for s in seeds]
        for a,b in pairs:
            require(a['evaluation_objects_identity']==b['evaluation_objects_identity'],'Budget comparison evaluation-object mismatch')
        eligible=all(a['eligible'] and b['eligible'] for a,b in pairs);metrics={}
        for metric in METRICS:
            deltas=[None if a['metrics_raw'][metric] is None or b['metrics_raw'][metric] is None else
                100*(b['metrics_raw'][metric]-a['metrics_raw'][metric]) for a,b in pairs]
            complete=eligible and all(v is not None for v in deltas)
            metrics[metric]=dict(deltas_pp=deltas,mean_pp=float(np.mean(deltas)) if complete else None,
                sd_pp=float(np.std(deltas,ddof=1)) if complete else None,eligible=complete,
                positive_seeds=sum(v>1e-10 for v in deltas) if complete else None,
                negative_seeds=sum(v < -1e-10 for v in deltas) if complete else None)
        out.append(dict(dataset=ds,method=method,seeds=seeds,from_Q=low,to_Q=high,metrics=metrics))
    return out

def plot_source_data(evidence):
    groups={}
    for key in ['main','original_budget_curves','new_repetition','ag_confirmation']:
        group=evidence[key];rows=[]
        for r in group['records']:
            row={name:r[name] for name in ['dataset','seed','Q','method','eligible','status','prediction_sha256','query_prefix_identity','evaluation_objects_identity','evaluation_pairs_identity']}
            row.update({m+'_percent':None if r['metrics_raw'][m] is None else 100*r['metrics_raw'][m] for m in METRICS});rows.append(row)
        groups[key]=dict(records_percent=rows,cells=group['cells'],paired_effects=group.get('paired',[]))
    groups['new_repetition']['budget_effects']=budget_effects(evidence['new_repetition'],MAIN_METHODS[:3])
    return dict(schema='sapn_v4_plot_sources/1',unit='percent for absolute metric columns; percentage points for differences',
        warning='Do not multiply *_percent or *_pp again. Groups overlap in recorded predictions and must not be pooled.',
        groups=groups,unified_status=evidence['unified_status'],dataset_names=DATASET_NAMES,method_names=METHOD_NAMES)


def write_outputs(result):
    tokens={k:v for k,v in result.items() if k!='_source_data'}
    (ROOT/'extension_results_tokens.json').write_text(json.dumps(tokens,ensure_ascii=False,indent=2),encoding='utf-8')
    (ROOT/'extension_plot_sources.json').write_text(json.dumps(result['_source_data'],ensure_ascii=False,indent=2),encoding='utf-8')
    records=[dict(group=group,**row) for group,data in result['_source_data']['groups'].items() for row in data['records_percent']]
    with (ROOT/'extension_all_records.csv').open('w',encoding='utf-8-sig',newline='') as stream:
        writer=csv.DictWriter(stream,fieldnames=list(records[0]));writer.writeheader();writer.writerows(records)
    manifest=dict(status='COMPLETE_VERIFIED_RESULTS_ONLY',source_hashes=result['_source_hashes'],
        outputs={name:sha(ROOT/name) for name in ['extension_results_tokens.json','extension_plot_sources.json','extension_all_records.csv']})
    (ROOT/'EXTENSION_RESULT_SOURCE_MANIFEST.json').write_text(json.dumps(manifest,indent=2),encoding='utf-8')
